    -- Lógica para detectar quando a bola está vindo em direção ao jogador
    -- e executar a ação de parry automaticamente
    
    -- Exemplo de implementação (precisa ser adaptado para o jogo específico)
    local ball = workspace:FindFirstChild("Ball")
    if ball and ball.Position then
        local distance = (ball.Position - LocalPlayer.Character.HumanoidRootPart.Position).Magnitude
        if distance < 10 then -- Distância de ativação
            -- Executar parry
            game:GetService("ReplicatedStorage"):WaitForChild("Events"):WaitForChild("Parry"):FireServer()
        end
    end
end

-- ESP
local function createESP()
    if not _G.Settings.ESP then return end
    
    -- Limpar ESPs anteriores
    for _, obj in pairs(workspace.CurrentCamera:GetChildren()) do
        if obj:IsA("BoxHandleAdornment") and obj.Name == "ESPBox" then
            obj:Destroy()
        end
    end
    
    -- Criar ESP para todos os jogadores
    for _, player in pairs(Players:GetPlayers()) do
        if player ~= LocalPlayer and player.Character and player.Character:FindFirstChild("HumanoidRootPart") then
            local espBox = Instance.new("BoxHandleAdornment")
            espBox.Name = "ESPBox"
            espBox.Parent = workspace.CurrentCamera
            espBox.Adornee = player.Character.HumanoidRootPart
            espBox.Size = player.Character.HumanoidRootPart.Size * Vector3.new(2, 2, 2)
            espBox.Color3 = player.Team and player.Team.TeamColor.Color or Color3.new(1, 0, 0)
            espBox.Transparency = 0.5
            espBox.ZIndex = 10
            espBox.AlwaysOnTop = true
        end
    end
end

-- Tracers
local function createTracers()
    if not _G.Settings.Tracers then return end
    
    -- Limpar Tracers anteriores
    for _, obj in pairs(workspace.CurrentCamera:GetChildren()) do
        if obj:IsA("Frame") and obj.Name == "TracerLine" then
            obj:Destroy()
        end
    end
    
    -- Criar Tracers para todos os jogadores
    for _, player in pairs(Players:GetPlayers()) do
        if player ~= LocalPlayer and player.Character and player.Character:FindFirstChild("HumanoidRootPart") then
            local tracer = Instance.new("Frame")
            tracer.Name = "TracerLine"
            tracer.Parent = workspace.CurrentCamera
            tracer.BackgroundColor3 = player.Team and player.Team.TeamColor.Color or Color3.new(1, 0, 0)
            tracer.BorderSizePixel = 0
            tracer.BackgroundTransparency = 0.5
            
            -- Atualizar posição do tracer a cada frame
            local connection
            connection = RunService.RenderStepped:Connect(function()
                if not _G.Settings.Tracers or not player.Character or not player.Character:FindFirstChild("HumanoidRootPart") then
                    tracer:Destroy()
                    connection:Disconnect()
                    return
                end
                
                local rootPos = player.Character.HumanoidRootPart.Position
                local screenPos = workspace.CurrentCamera:WorldToScreenPoint(rootPos)
                
                if screenPos.Z > 0 then
                    tracer.Visible = true
                    tracer.Position = UDim2.new(0, screenPos.X, 0, screenPos.Y)
                    tracer.Size = UDim2.new(0, 1, 0, workspace.CurrentCamera.ViewportSize.Y - screenPos.Y)
                    tracer.Rotation = math.atan2(workspace.CurrentCamera.ViewportSize.X/2 - screenPos.X, workspace.CurrentCamera.ViewportSize.Y/2 - screenPos.Y) * (180/math.pi)
                else
                    tracer.Visible = false
                end
            end)
        end
    end
end

-- Chams
local function createChams()
    if not _G.Settings.Chams then return end
    
    -- Limpar Chams anteriores
    for _, player in pairs(Players:GetPlayers()) do
        if player ~= LocalPlayer and player.Character then
            for _, part in pairs(player.Character:GetChildren()) do
                if part:IsA("BasePart") and part:FindFirstChild("Cham") then
                    part.Cham:Destroy()
                end
            end
        end
    end
    
    -- Criar Chams para todos os jogadores
    for _, player in pairs(Players:GetPlayers()) do
        if player ~= LocalPlayer and player.Character then
            for _, part in pairs(player.Character:GetChildren()) do
                if part:IsA("BasePart") then
                    local cham = Instance.new("BoxHandleAdornment")
                    cham.Name = "Cham"
                    cham.Parent = part
                    cham.Adornee = part
                    cham.Size = part.Size
                    cham.Color3 = player.Team and player.Team.TeamColor.Color or Color3.new(1, 0, 0)
                    cham.Transparency = 0.3
                    cham.ZIndex = 5
                    cham.AlwaysOnTop = true
                end
            end
        end
    end
end

-- Ball ESP
local function createBallESP()
    if not _G.Settings.BallESP then return end
    
    -- Limpar Ball ESPs anteriores
    local ball = workspace:FindFirstChild("Ball")
    if ball and ball:FindFirstChild("BallESP") then
        ball.BallESP:Destroy()
    end
    
    -- Criar ESP para a bola
    if ball then
        local ballEsp = Instance.new("BoxHandleAdornment")
        ballEsp.Name = "BallESP"
        ballEsp.Parent = ball
        ballEsp.Adornee = ball
        ballEsp.Size = ball.Size * Vector3.new(1.5, 1.5, 1.5)
        ballEsp.Color3 = Color3.new(1, 1, 0) -- Amarelo para destacar
        ballEsp.Transparency = 0.3
        ballEsp.ZIndex = 10
        ballEsp.AlwaysOnTop = true
    end
end

-- No Clip
local function noClip()
    if not _G.Settings.NoClip then return end
    
    if LocalPlayer.Character then
        for _, part in pairs(LocalPlayer.Character:GetChildren()) do
            if part:IsA("BasePart") then
                part.CanCollide = false
            end
        end
    end
end

-- Infinite Jump
local function infiniteJump()
    if not _G.Settings.InfiniteJump then return end
    
    UserInputService.JumpRequest:Connect(function()
        if LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("Humanoid") then
            LocalPlayer.Character.Humanoid.Jump = true
        end
    end)
end

-- Speed Boost
local function speedBoost()
    if not _G.Settings.SpeedBoost or _G.Settings.SpeedBoost == 16 then return end
    
    if LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("Humanoid") then
        LocalPlayer.Character.Humanoid.WalkSpeed = _G.Settings.SpeedBoost
    end
end

-- Fly
local flyEnabled = false
local flySpeed = 50

local function fly()
    if not _G.Settings.Fly then 
        flyEnabled = false
        return 
    end
    
    flyEnabled = true
    
    local character = LocalPlayer.Character
    local humanoid = character and character:FindFirstChildOfClass("Humanoid")
    local rootPart = character and character:FindFirstChild("HumanoidRootPart")
    
    if not humanoid or not rootPart then return end
    
    local bv = Instance.new("BodyVelocity")
    bv.MaxForce = Vector3.new(math.huge, math.huge, math.huge)
    bv.P = 5000
    bv.Parent = rootPart
    
    local bg = Instance.new("BodyGyro")
    bg.MaxTorque = Vector3.new(math.huge, math.huge, math.huge)
    bg.P = 5000
    bg.Parent = rootPart
    
    local connection
    connection = RunService.RenderStepped:Connect(function()
        if not _G.Settings.Fly or not character or not humanoid or not rootPart then
            bv:Destroy()
            bg:Destroy()
            connection:Disconnect()
            return
        end
        
        humanoid.PlatformStand = true
        
        local moveDirection = humanoid.MoveDirection
        if moveDirection ~= Vector3.new(0, 0, 0) then
            bv.Velocity = moveDirection * flySpeed
        else
            bv.Velocity = Vector3.new(0, 0, 0)
        end
        
        bg.CFrame = workspace.CurrentCamera.CFrame
    end)
end
